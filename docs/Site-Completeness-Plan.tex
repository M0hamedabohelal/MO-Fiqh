\section{الهدف}

\begin{goalbox}
سدّ الفجوات التي تفصل منصة يَسِّر — المكتملة وظيفياً وتقنياً — عن موقع جاهز
للجمهور العام والتطبيقات: \textbf{الصفحات القانونية}، \textbf{صفحة عن المنصة}،
\textbf{تحليلات GA4}، و\textbf{تذكير القراءة المحلي}.
\end{goalbox}

\subsection*{نتائج الفحص المُوجِزة}
\begin{itemize}
  \item لا توجد أي صفحات قانونية (خصوصية · شروط · إخلاء · حقوق).
  \item لا توجد صفحة «عن المنصة والمنهجية».
  \item لا تحليلات ولا تحقق Google Search Console في \texttt{index.html}.
  \item لا إشعارات تذكير رغم وجود سلاسل مواظبة.
  \item \texttt{prefers-color-scheme} غير مطبّق (الثيم داكن دائماً).
\end{itemize}

\section{المرحلة ١ — الصفحات القانونية}

\subsection*{المخرَج: أربع صفحات ثابتة}
تُبنى جميعها عبر \texttt{scripts/generate-seo.mjs} بالقالب الموجود (\texttt{head()} / \texttt{foot()})
في مسارات نظيفة بنهاية مائلة.

\vspace{6pt}
\renewcommand{\arraystretch}{1.35}
\begin{longtable}{@{}p{0.20\textwidth} p{0.72\textwidth}@{}}
\toprule
\textbf{الصفحة} & \textbf{المحتوى} \\
\midrule
\endhead
\texttt{/privacy/} & ما يُجمَع (Auth · مكتبة المستخدم · إحصاءات مجهولة · GA4) · أين يُخزَّن (Firestore + محلي) · مدة الحفظ · حقّ النسيان (منفّذ عبر حذف المكتبة) \\
\texttt{/terms/} & شروط الاستخدام · مسؤولية المستخدم · حدود الخدمة \\
\texttt{/disclaimer/} & إخلاء المسؤولية · المحتوى اجتهاد فقهي يُراجَع مع أهل العلم \\
\texttt{/copyright/} & حقوق النشر · نسبة المحتوى للشيخ · ترخيص الاستخدام \\
\bottomrule
\end{longtable}
\renewcommand{\arraystretch}{1.0}

\subsubsection*{التنفيذ}
\begin{itemize}
  \item دالة جديدة \texttt{legalPage(slug, title, bodyHtml)} في \texttt{generate-seo.mjs}.
  \item مصفوفة صفحات قانونية تُبَنى تلقائياً مع كل بناء، وتُضاف إلى \texttt{sitemap.xml}.
  \item ربط الصفحات في فوتر \texttt{foot()} لكل الصفحات الثابتة.
  \item إضافة روابط في فوتر التطبيق التفاعلي (شاشة الإعدادات).
\end{itemize}

\subsection*{التحقق}
توسيع \texttt{check-seo.mjs} ليفشل صراحةً عند غياب أي صفحة قانونية أو رابطها.

\begin{warnbox}
\textbf{قاعدة حاكمة:} سياسة الخصوصية يجب أن تذكر \textbf{GA4} صراحةً (ملفات تعريف ارتباط / قياس)
قبل إضافته — وإلا وقع تناقض قانوني. لذلك المرحلة ١ تسبق المرحلة ٣.
\end{warnbox}

\section{المرحلة ٢ — صفحة «عن المنصة والمنهجية»}

\begin{itemize}
  \item \textbf{المسار:} \texttt{/about/}.
  \item \textbf{المحتوى:} الرسالة · المنهجية (الكتاب والسنة) · مصدر المحتوى (فضيلة الشيخ) · آلية التحقق والمراجعة.
  \item \textbf{الأثر على SEO:} تقوية \textbf{E-E-A-T} (الخبرة · التخصص · الموثوقية) — عامل حاسم لجوجل في المحتوى الديني.
  \item \textbf{التنفيذ:} دالة \texttt{aboutPage()} + ربط في الفوتر + \texttt{AboutPage} schema.
\end{itemize}

% ============================================================
\section{المرحلة ٣ — تحليلات GA4}

\subsection*{التنفيذ}
\begin{itemize}
  \item سكربت GA4 في \texttt{index.html} بعد موافقة لطيفة (بلا تخزين مسبق).
  \item قياس انتقالات التطبيق الهاشيّ (\texttt{useHashRoute}) كأحداث \texttt{page\_view} يدوية.
  \item ملف جديد \texttt{src/utils/analytics.ts}: غلاف آمن (لا يكسر التطبيق إن فشل التحميل).
\end{itemize}

\begin{warnbox}
\textbf{قيد CSP إلزامي:} يجب تعديل \texttt{firebase.json} وإلا حُجب GA4 \textbf{صامتاً}:
\begin{codebox}
{\ttfamily\small\raggedleft
script-src \ ...\ https://www.googletagmanager.com;\\
connect-src \ ...\ https://*.google-analytics.com\ https://*.analytics.google.com;
}
\end{codebox}
هذه نقطة نسيان شائعة موثّقة سابقاً في مشروع المنصة.
\end{warnbox}

\subsection*{كيف نتحقق من عملها}
فتح لوحة GA4 ← ``Realtime'' ← التصفح يظهر خلال ثوانٍ · ثم فحص Console بلا أخطاء CSP.

\section{المرحلة ٤ — تذكير القراءة المحلي}

\begin{notebox}
\textbf{القرار:} تذكير محلي فقط عبر \texttt{Notification API} — بلا FCM ولا خادم،
يناسب نطاق النشر الحالي (Firebase Hosting بلا Functions).
\end{notebox}

\begin{itemize}
  \item طلب إذن الإشعارات مرة واحدة بلطف (لكل مستخدم).
  \item جدولة تذكير يومي داخل التطبيق مرتبط بسلسلة المواظبة.
  \item ملف جديد \texttt{src/hooks/useReadingReminder.ts}.
  \item \textbf{الحد المعلوم:} الإشعار يظهر عند فتح التطبيق/التصفح (لا خادم يوقظه مغلقاً).
\end{itemize}

% ============================================================
\section{المرحلة ٥ — تحسينات إكمال}

\begin{itemize}
  \item احترام \texttt{prefers-color-scheme} عند أول زيارة (الثيم الافتراضي يتبع النظام، والحفظ اليدوي يبقى).
  \item توسيع \texttt{robots.txt} + إشارة الصفحات القانونية للزاحف.
  \item مراجعة صفحة تواصل عامة (خارج شاشة الإعدادات).
\end{itemize}

\section{الملفات والتحقق}

\textbf{ملفات تُعدَّل/تُضاف:}
\begin{itemize}
  \item \texttt{scripts/generate-seo.mjs} — دوال الصفحات القانونية و«عن المنصة».
  \item \texttt{scripts/check-seo.mjs} — فحص وجود الصفحات القانونية.
  \item \texttt{index.html} — سكربت GA4 + رابط الفوتر.
  \item \texttt{firebase.json} — CSP لـ GA4.
  \item \texttt{src/utils/analytics.ts} · \texttt{src/hooks/useReadingReminder.ts} (جديد).
\end{itemize}
