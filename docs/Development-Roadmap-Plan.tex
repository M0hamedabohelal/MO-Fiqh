\section{الموجة الأولى: الأداء والأمان}

\subsection{البند 3 — مشاركة الفوائد كصورة محسّنة}

\begin{notebox}
\textbf{الحالة:} يوجد \texttt{QuoteImageModal.tsx} و\texttt{ShareButton.tsx} ومكتبة \texttt{html-to-image}
جاهزة. البند = رفع الجودة لا بناءً من الصفر.
\end{notebox}

\begin{itemize}
  \item \textbf{قوالب متعددة:} ٣ قوالب (كلاسيكي ذهبي، داكن عصري، ورقي تراثي) يُختار بينها قبل التصدير.
  \item \textbf{نِسَب جاهزة:} مربع ١:١ (إنستغرام/فيسبوك)، طولي ٩:١٦ (ستوريز)، أفقي ١٦:٩ (تويتر).
  \item \textbf{علامة مائية موحّدة:} شعار + «fqh.me» بموضع ثابت وزاوية خفيفة.
  \item \textbf{مشاركة مباشرة:} استخدام Web Share API (\texttt{navigator.share}) لملف الصورة عند الدعم،
        و\texttt{fallback} تنزيل.
  \item \textbf{الملفات:} \texttt{QuoteImageModal.tsx} (القوالب والنِسَب)، \texttt{ShareButton.tsx} (المشاركة المباشرة).
  \item \textbf{التحقق:} اختبار وحدة لتفاوت القوالب + فحص بصري يدوي.
\end{itemize}

\subsection{البند 5 — وضع المراجعة والاختبار السريع}

\begin{itemize}
  \item \textbf{مصدر البيانات:} المسائل المقروءة (\texttt{readLessons}) والفوائد المحفوظة.
  \item \textbf{«اختبار سريع» في شاشة القراءة:} زر يخفي التعليق ويطلب من المستخدم استذكار الحكم، ثم كشف.
  \item \textbf{«مراجعة الفوائد»:} شاشة بطاقات (\texttt{HighlightsView}) بوضع قلب البطاقة (سؤال ← جواب).
  \item \textbf{تتبّع التقدّم:} دمج نتيجة المراجعة في \texttt{AchievementsView} (سلسلة مواظبة/إتمام).
  \item \textbf{الملفات:} مكوّن جديد \texttt{ReviewMode.tsx} + hook \texttt{useReviewMode.ts}، وربط بالـ \texttt{App.tsx}.
  \item \textbf{التحقق:} اختبارات وحدة للمنطق + مسار e2e.
\end{itemize}

\subsection{البند 7 — \texttt{content-visibility} للأقسام الطويلة}

\begin{itemize}
  \item تطبيق \texttt{content-visibility: auto} + \texttt{contain-intrinsic-size} على قوائم المسائل
        وبطاقات الشرح في \texttt{LessonsView}/\texttt{ReadingView}.
  \item \textbf{الملفات:} \texttt{index.css} + \texttt{exterior.css} (أصناف مخصصة موجودة).
  \item \textbf{الحذر:} \texttt{contain-intrinsic-size} بقيمة واقعية لمنع قفزات التمرير.
  \item \textbf{التحقق:} قياس زمن التمرير قبل/بعد + عدم ظهور قفزات.
\end{itemize}

\subsection{البند 8 — استيراد Bootstrap مُجزَّأ}

\begin{notebox}
\textbf{الحالة:} \texttt{main.tsx} يستورد \texttt{bootstrap.rtl.min.css} كاملة (٢٢٥KB، ٣١KB gzip)
بينما المستخدم أجزاء قليلة (grid، utilities، reset).
\end{notebox}

\begin{itemize}
  \item \textbf{الخطوة ١:} جرد أصناف Bootstrap المستخدمة فعليًا (\texttt{rg} على \texttt{className}).
  \item \textbf{الخطوة ٢:} استيراد ملفات SCSS الجزئية فقط من \texttt{bootstrap/scss} بدل الحزمة الكاملة
        (\texttt{\_grid}, \texttt{\_utilities}, \texttt{\_reboot}, \texttt{\_buttons}, \texttt{\_forms}).
  \item \textbf{الخطوة ٣:} ضبط Vite على دعم Sass (\texttt{sass} كـ devDependency).
  \item \textbf{البديل الأبسط:} PurgeCSS على المخرج إن أردنا تجنّب SCSS.
  \item \textbf{التحقق:} مقارنة حجم CSS النهائي (المتوقّع توفير ٨٠--١٢٠KB) + فحص بصري لكل الصفحات.
\end{itemize}

\subsection{البند 9 — تفعيل App Check}

\begin{warnbox}
\textbf{تحذير حرج:} \texttt{initAppCheck} موجود لكنه مُعطَّل. التفعيل **قبل** تسجيل التطبيق في الكونسول
يُنتج خطأ ٤٠٠ في كل تحميل بلا أي حماية. التسلسل إلزامي.
\end{warnbox}

\begin{enumerate}
  \item تسجيل تطبيق الويب في Firebase Console ← App Check ← reCAPTCHA v3، واستخراج المفتاح.
  \item خبز \texttt{VITE\_FIREBASE\_APPCHECK\_KEY} في بناء الإنتاج، و\texttt{VITE\_ENABLE\_APPCHECK=true}.
  \item مراقبة لوحة App Check حتى تصبح الطلبات «متحقّقة» بنسبة عالية.
  \item \textbf{بعدها فقط:} إنفاذ \texttt{appCheck.token} في \texttt{firestore.rules}.
  \item \textbf{الملفات:} \texttt{src/firebase/config.ts} (موجود)، \texttt{firestore.rules}، إعدادات النشر.
  \item \textbf{التحقق:} تتبّع نسبة التحقق في الكونسول + عدم كسر كتابات العملاء.
\end{enumerate}

\subsection{البند 10 — الرفع الموقّع لـ Cloudinary}

\begin{warnbox}
\textbf{الحالة الأمنية:} الرفع الحالي \textbf{غير موقّع} من العميل — أي زائر يستهلك حصّة الرفع
(\texttt{utils/cloudinary.ts}). هذا إصلاح أمني حقيقي لا تحسين تجميلي.
\end{warnbox}

\begin{itemize}
  \item \textbf{النسخة الكاملة:} Cloud Function (تُضاف لاحقًا) تولّد توقيعًا لحظيًا؛
        \texttt{CLOUDINARY\_API\_SECRET} في Secret Manager فقط.
  \item \textbf{النسخة الوسيطة (بلا سيرفر):} تقييد الـ preset في Cloudinary أشدّ: تنسيقات مسموحة،
        حدّ حجم، مجلد ثابت، ورفع \texttt{moderation} — تقليل سطح العبث فورًا.
  \item \textbf{الملفات:} \texttt{utils/cloudinary.ts} (طلب التوقيع + استخدامه)، وإعداد Cloudinary.
  \item \textbf{التحقق:} \texttt{grep -r "CLOUDINARY\_API\_SECRET\|AIza" dist/} ← لا نتائج.
\end{itemize}

\subsection{البند 12 — اختبار قواعد Firestore عبر Emulator}

\begin{notebox}
\textbf{الحالة:} \texttt{rulesSchema.test.ts} الحالي هو فحص \textbf{تطابق نصوصي} (contract check) لا تشغيل حقيقي.
\end{notebox}

\begin{itemize}
  \item إضافة \texttt{@firebase/rules-unit-testing} كـ devDependency.
  \item كتابة اختبارات تُشغّل Emulator وتثبّت: قراءة عامة للمحتوى، رفض كتابة العميل على \texttt{lessons}،

  \item \textbf{الملفات:} \texttt{src/firebase/rules.test.ts} جديد + سكربت \texttt{test:rules}.
  \item \textbf{التحقق:} تشغيل محلي في CI (يحتاج Java للـ Emulator).
\end{itemize}

\subsection{البند 13 — تقسيم الملفات العملاقة}

\begin{itemize}
  \item \texttt{AdminLessonsTab.tsx} (٤٠KB): استخراج مكوّنات فرعية (نموذج التحرير، قائمة المسائل، سلة المحذوفات).
  \item \texttt{SettingsView.tsx} (٤٠KB): استخراج \texttt{SettingsSection} و\texttt{Feedback} لمكوّنات مستقلة.
  \item \textbf{الهدف:} كل ملف < ٢٠KB، ومسؤولية واحدة واضحة.
  \item \textbf{التحقق:} \texttt{typecheck} + \texttt{lint} + الاختبارات القائمة تبقى خضراء (لا تغيير سلوكي).

\subsection{البند 14 — توسيع اختبارات e2e}

\begin{notebox}
\textbf{الحالة:} ١١ اختبار e2e تغطي المسارات الحرجة. البند = تغطية الشاشات المتبقية.
\end{notebox}

\begin{itemize}
  \item مسارات جديدة: الإعدادات (فتح/طي الـ Accordion)، الإنجازات، الفوائد المقتبسة، الوضع الداكن/سيپيا،

  \item \textbf{الملفات:} \texttt{e2e/app.e2e.ts} (إضافة describe جديدة).
  \item \textbf{التحقق:} تشغيل \texttt{npm run test:e2e} أخضر مع الميزانية الحالية (\texttt{workers: 1}).

\subsection{البند 15 — إثراء Schema.org}

\begin{notebox}
\textbf{الحالة:} \texttt{Article} و\texttt{BreadcrumbList} و\texttt{WebSite} و\texttt{Organization}
موجودة بالفعل في \texttt{scripts/generate-seo.mjs}. البند = إضافة أنواع جديدة.

\begin{itemize}
  \item \textbf{FAQPage:} لكل مسألة، بناء أسئلة من العنوان + الحكم (يحسّن ظهور النتائج الغنية).
  \item \textbf{QAPage:} البديل الأنسب للمحتوى الفقهي سؤال/جواب.

  \item \textbf{التحقق:} \texttt{check:seo} + Rich Results Test من جوجل.
\end{itemize}

\subsection{البند 16 — sitemap ديناميكي مع lastmod}

\begin{itemize}
  \item إضافة \texttt{<lastmod>} لكل مسألة من طابع تعديلها (يُخزَّن عند النشر).
  \item التأكد أن \texttt{sitemap.xml} يُحدَّث آليًا مع كل مسألة جديدة (مولّد أصلًا في البناء).
  \item \textbf{الملفات:} \texttt{scripts/generate-seo.mjs} (قسم الـ sitemap).
  \item \textbf{التحقق:} \texttt{check:seo} + رفع الموقع في Search Console.
\end{itemize}
